cmake_minimum_required(VERSION 3.16)
project(sin_episodes_quest3_compat VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
add_library(sin_weapon_runtime STATIC src/weapon_runtime.cpp)
target_include_directories(sin_weapon_runtime PUBLIC include third_party/research)
target_compile_options(sin_weapon_runtime PRIVATE $<$<CXX_COMPILER_ID:GNU,Clang>:-Wall;-Wextra;-Wpedantic>)
# Load script settings from the owner's own game installation.
add_library(sin_weapon_scripts STATIC src/weapon_scripts.cpp)
target_include_directories(sin_weapon_scripts PUBLIC include)
target_link_libraries(sin_weapon_scripts PUBLIC sin_weapon_runtime)
target_compile_options(sin_weapon_scripts PRIVATE $<$<CXX_COMPILER_ID:GNU,Clang>:-Wall;-Wextra;-Wpedantic>)
add_executable(sin_weapon_script_check tools/sin_weapon_script_check.cpp)
target_link_libraries(sin_weapon_script_check PRIVATE sin_weapon_scripts)
include(CTest)
if(BUILD_TESTING)
  add_executable(sin_weapon_runtime_tests tests/test_weapon_runtime.cpp)
  target_link_libraries(sin_weapon_runtime_tests PRIVATE sin_weapon_runtime)
  add_test(NAME sin_weapon_runtime_tests COMMAND sin_weapon_runtime_tests)
  add_executable(sin_weapon_scripts_tests tests/test_weapon_scripts.cpp)
  target_link_libraries(sin_weapon_scripts_tests PRIVATE sin_weapon_scripts)
  add_test(NAME sin_weapon_scripts_tests COMMAND sin_weapon_scripts_tests)
endif()
