cmake_minimum_required(VERSION 3.16)
project(sin_episodes_quest3_compat VERSION 0.1.0 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
add_library(sin_weapon_runtime STATIC src/weapon_runtime.cpp)
target_include_directories(sin_weapon_runtime PUBLIC include third_party/research)
set_target_properties(sin_weapon_runtime PROPERTIES POSITION_INDEPENDENT_CODE ON)
target_compile_options(sin_weapon_runtime PRIVATE $<$<CXX_COMPILER_ID:GNU,Clang>:-Wall;-Wextra;-Wpedantic>)
# This is our own host-neutral C ABI, not the Source Engine CreateInterface ABI.
add_library(sin_weapon_bridge SHARED src/weapon_c_bridge.cpp)
target_include_directories(sin_weapon_bridge PUBLIC include)
target_link_libraries(sin_weapon_bridge PRIVATE sin_weapon_runtime)
target_compile_options(sin_weapon_bridge PRIVATE $<$<CXX_COMPILER_ID:GNU,Clang>:-Wall;-Wextra;-Wpedantic>)
set_target_properties(sin_weapon_bridge PROPERTIES CXX_VISIBILITY_PRESET hidden VISIBILITY_INLINES_HIDDEN YES)
include(CTest)
if(BUILD_TESTING)
  add_executable(sin_weapon_runtime_tests tests/test_weapon_runtime.cpp)
  target_link_libraries(sin_weapon_runtime_tests PRIVATE sin_weapon_runtime)
  add_test(NAME sin_weapon_runtime_tests COMMAND sin_weapon_runtime_tests)
  add_executable(sin_weapon_bridge_tests tests/test_weapon_c_bridge.c)
  target_link_libraries(sin_weapon_bridge_tests PRIVATE sin_weapon_bridge)
  set_target_properties(sin_weapon_bridge_tests PROPERTIES LINKER_LANGUAGE CXX)
  add_test(NAME sin_weapon_bridge_tests COMMAND sin_weapon_bridge_tests)
endif()
