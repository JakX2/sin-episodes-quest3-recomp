cmake_minimum_required(VERSION 3.16)
project(sin_episodes_quest3_compat VERSION 0.1.0 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
add_library(sin_weapon_runtime STATIC src/weapon_runtime.cpp)
target_include_directories(sin_weapon_runtime PUBLIC include third_party/research)
target_compile_options(sin_weapon_runtime PRIVATE $<$<CXX_COMPILER_ID:GNU,Clang>:-Wall;-Wextra;-Wpedantic>)
# R1-only diagnostic factory; deliberately does NOT implement Source game APIs.
add_library(sin_factory_probe SHARED src/sourcevr_factory_probe.c)
target_compile_options(sin_factory_probe PRIVATE $<$<C_COMPILER_ID:GNU,Clang>:-Wall;-Wextra;-Wpedantic>)
if(ANDROID)
  target_link_libraries(sin_factory_probe PRIVATE log)
endif()
include(CTest)
if(BUILD_TESTING)
  add_executable(sin_weapon_runtime_tests tests/test_weapon_runtime.cpp)
  target_link_libraries(sin_weapon_runtime_tests PRIVATE sin_weapon_runtime)
  add_test(NAME sin_weapon_runtime_tests COMMAND sin_weapon_runtime_tests)
  if(UNIX AND NOT ANDROID)
    add_executable(sin_factory_probe_tests tests/test_sourcevr_factory_probe.c)
    target_link_libraries(sin_factory_probe_tests PRIVATE ${CMAKE_DL_LIBS})
    add_test(NAME sin_factory_probe_tests
             COMMAND sin_factory_probe_tests $<TARGET_FILE:sin_factory_probe>)
  endif()
endif()
